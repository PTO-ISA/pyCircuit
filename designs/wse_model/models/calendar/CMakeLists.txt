cmake_minimum_required(VERSION 3.25)
project(WseCalendar LANGUAGES CXX)

find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(pycircuit CONFIG REQUIRED COMPONENTS Runtime)
include("${PYCIRCUIT_CMAKE_DIR}/PycircuitExamples.cmake")

set(_suite "$ENV{WSE_CALENDAR_SUITE}")
if(NOT _suite OR _suite STREQUAL "route")
  set(_top RouteSystem)
  set(_entry bench_route)
  set(_sources route.py bench_route.py)
  set(_cycles 4)
elseif(_suite STREQUAL "receive")
  set(_top ReceiveSystem)
  set(_entry bench_receive)
  set(_sources receive_checks.py receive.py bench_receive.py)
  set(_cycles 12)
elseif(_suite STREQUAL "control")
  set(_top ControlSystem)
  set(_entry bench_control)
  set(_sources control.py bench_control.py)
  set(_cycles 261)
elseif(_suite STREQUAL "forward")
  set(_top ForwardSystem)
  set(_entry bench_forward)
  set(_sources forward.py bench_forward.py)
  set(_cycles 9)
elseif(_suite STREQUAL "multinode")
  set(_top MultiNodeSystem)
  set(_entry bench_multinode)
  set(_sources forward.py receive_checks.py receive.py bench_multinode.py)
  set(_cycles 12)
elseif(_suite STREQUAL "allgather")
  set(_top AllGatherSystem)
  set(_entry bench_allgather)
  set(_sources forward.py receive_checks.py receive.py allgather.py bench_allgather.py)
  set(_cycles 516)
elseif(_suite STREQUAL "allgather-control" OR _suite STREQUAL "allgather-delayed")
  set(_top AllGatherControlSystem)
  set(_cycles 523)
  if(_suite STREQUAL "allgather-delayed")
    set(_top AllGatherDelayedSystem)
    set(_cycles 529)
  endif()
  set(_entry bench_allgather_control)
  set(_sources control.py route.py forward.py receive_checks.py receive.py
      allgather.py allgather_control.py bench_allgather_control.py)
elseif(_suite STREQUAL "node")
  set(_top NodeSystem)
  set(_entry bench_node)
  set(_sources control.py route.py forward.py receive_checks.py receive.py bench_node.py)
  set(_cycles 13)
elseif(_suite STREQUAL "capacity-positive" OR _suite STREQUAL "capacity-exhausted")
  if(_suite STREQUAL "capacity-positive")
    set(_top CapacityAvailableSystem)
  else()
    set(_top CapacityExhaustedSystem)
  endif()
  set(_entry bench_capacity)
  set(_sources receive_checks.py bench_capacity.py)
  set(_cycles 3)
else()
  message(FATAL_ERROR "Unknown WSE_CALENDAR_SUITE: ${_suite}")
endif()

pycircuit_add_example(calendar_${_suite}
  PACKAGE wse_calendar
  TOP wse_calendar.${_entry}.${_top}
  ENTRY_SOURCE ${_entry}
  SOURCES ${_sources}
  CYCLES ${_cycles})
