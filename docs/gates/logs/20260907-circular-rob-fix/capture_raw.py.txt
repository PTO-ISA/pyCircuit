import sys
from pathlib import Path
if len(sys.argv) > 1:
    sys.path.insert(0, str(Path(sys.argv[1]).resolve()))
from agentic_circuit._queue_frontend import lower_queue_source
out = Path('.pycircuit_out/rob-regression/raw-before' if len(sys.argv) > 1 else '.pycircuit_out/rob-regression/raw-after')
out.mkdir(exist_ok=True)
for name in ('circular_rob', 'reusable_circular_rob'):
    source = Path('examples/agentic-circuit/state') / (name + '.py')
    (out / (name+'.mlir')).write_text(lower_queue_source(source.read_text(), name))
