import importlib.util
import tempfile
import unittest
from pathlib import Path
spec = importlib.util.spec_from_file_location('queue_test', 'tests/integration/agentic-circuit/e2e/test_queue_codegen.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
class Retain:
    def __enter__(self):
        self.path = tempfile.mkdtemp(dir=Path('.pycircuit_out/rob-regression').resolve())
        print(self.path, flush=True)
        return self.path
    def __exit__(self, *args): pass
m.tempfile.TemporaryDirectory = Retain
suite = unittest.TestSuite(m.QueueCodegenTest(n) for n in [
 'test_circular_rob_wraps_rejects_stale_completion_and_recovers',
 'test_reusable_circular_rob_preserves_rules_and_instance_state',
 'test_reusable_rob_scan_and_activation_match_every_tick'])
unittest.TextTestRunner(verbosity=2).run(suite)
