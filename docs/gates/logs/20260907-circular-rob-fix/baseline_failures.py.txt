import importlib.util
import shutil
import subprocess
import unittest
from pathlib import Path
root = Path.cwd()
out = root / '.pycircuit_out/rob-regression/baseline-python'
shutil.copytree(root / 'python/agentic-circuit/src/agentic_circuit', out / 'agentic_circuit', dirs_exist_ok=True, ignore=shutil.ignore_patterns('__pycache__'))
original = subprocess.check_output(['git', 'show', 'HEAD:python/agentic-circuit/src/agentic_circuit/_queue_frontend.py'])
(out / 'agentic_circuit/_queue_frontend.py').write_bytes(original)
spec = importlib.util.spec_from_file_location('queue_test', root / 'tests/integration/agentic-circuit/e2e/test_queue_codegen.py')
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
run = subprocess.run
def baseline_run(*args, **kwargs):
    env = kwargs.get('env')
    if env and 'PYTHONPATH' in env:
        kwargs['env'] = {**env, 'PYTHONPATH': str(out) + ':' + env['PYTHONPATH']}
    return run(*args, **kwargs)
m.subprocess.run = baseline_run
unittest.TextTestRunner(verbosity=2).run(unittest.TestSuite(m.QueueCodegenTest(n) for n in (
 'test_reusable_oldest_ready_isq_closes_lost_wakeup_and_backpressure',
 'test_same_owner_branch_join_emits_one_state_proposal')))
