cmake_minimum_required(VERSION 3.20)
project(gfsim VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(GFSIM_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(GFSIM_BUILD_EXAMPLES "Build circuit runners and benchmarks" ON)
add_library(gfsim src/simulator.cpp)
add_library(gfsim::gfsim ALIAS gfsim)
target_compile_features(gfsim PUBLIC cxx_std_20)
target_include_directories(gfsim PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_options(gfsim PRIVATE -Wall -Wextra -Wpedantic)
if(GFSIM_SANITIZERS)
  # Keep debug symbols with modest optimization for the 1100-stage instrumented drain.
  target_compile_options(gfsim PUBLIC -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gfsim PUBLIC -fsanitize=address,undefined)
endif()
install(TARGETS gfsim EXPORT gfsimTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT gfsimTargets NAMESPACE gfsim:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gfsim)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/gfsimConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/gfsimConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gfsim)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/gfsimConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/gfsimConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/gfsimConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/gfsim)
if(GFSIM_BUILD_EXAMPLES)
  add_executable(gfsim-ripes5 examples/ripes5/tests/runner.cpp)
  target_link_libraries(gfsim-ripes5 PRIVATE gfsim::gfsim)
  target_compile_options(gfsim-ripes5 PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_executable(gfsim-native-tests tests/native.cpp tests/circuits/pipeline/test.cpp tests/semantics.cpp)
  target_link_libraries(gfsim-native-tests PRIVATE gfsim::gfsim)
  add_test(NAME native-components-and-boundaries COMMAND gfsim-native-tests)
  set_tests_properties(native-components-and-boundaries PROPERTIES TIMEOUT 300)
  add_executable(gfsim-circuits tests/circuits/runner.cpp)
  target_link_libraries(gfsim-circuits PRIVATE gfsim::gfsim)
  add_executable(gfsim-equivalence tests/circuits/equivalence.cpp)
  target_link_libraries(gfsim-equivalence PRIVATE gfsim::gfsim)
  add_test(NAME circuit-equivalence COMMAND gfsim-equivalence)
  configure_file(tests/install.cmake.in install-test.cmake @ONLY)
  add_test(NAME install-and-link COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_BINARY_DIR}/install-test.cmake)
endif()
if(BUILD_TESTING AND GFSIM_BUILD_EXAMPLES)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(GFSIM_RIPES_REFERENCE "${CMAKE_CURRENT_SOURCE_DIR}/../../reference/builds/ripes-reference/ripes5-reference" CACHE FILEPATH "Pinned native Ripes reference, required for formal acceptance")
  add_test(NAME ripes5-three-way COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/examples/ripes5/tests/verify.py
    --cpp-runner $<TARGET_FILE:gfsim-ripes5> --runner ${GFSIM_RIPES_REFERENCE}
    --output ${CMAKE_CURRENT_BINARY_DIR}/ripes5-evidence)
  set_tests_properties(ripes5-three-way PROPERTIES TIMEOUT 300)
  add_test(NAME runner-input-and-reference-gates COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_input.py $<TARGET_FILE:gfsim-ripes5>)
endif()
