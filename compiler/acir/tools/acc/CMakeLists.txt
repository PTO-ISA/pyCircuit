add_executable(acc acc.cpp)

file(STRINGS "${ACIR_REPO_ROOT}/pyproject.toml" ACC_PRODUCT_VERSION_LINE
  REGEX "^version = \"[0-9]+\\.[0-9]+\\.[0-9]+\"$"
  LIMIT_COUNT 1
)
string(REGEX REPLACE "^version = \"([^\"]+)\"$" "\\1"
  ACC_PRODUCT_VERSION "${ACC_PRODUCT_VERSION_LINE}"
)

set(ACC_SOURCE_REVISION "${PYC_TOOLCHAIN_GIT_SHA}")
string(LENGTH "${ACC_SOURCE_REVISION}" ACC_SOURCE_REVISION_LENGTH)
if(NOT ACC_SOURCE_REVISION_LENGTH EQUAL 40 OR
   NOT ACC_SOURCE_REVISION MATCHES "^[0-9a-f]+$")
  execute_process(
    COMMAND git rev-parse HEAD
    WORKING_DIRECTORY "${ACIR_REPO_ROOT}"
    OUTPUT_VARIABLE ACC_SOURCE_REVISION
    OUTPUT_STRIP_TRAILING_WHITESPACE
    ERROR_QUIET
  )
endif()
string(LENGTH "${ACC_SOURCE_REVISION}" ACC_SOURCE_REVISION_LENGTH)
if(NOT ACC_SOURCE_REVISION_LENGTH EQUAL 40 OR
   NOT ACC_SOURCE_REVISION MATCHES "^[0-9a-f]+$")
  set(ACC_SOURCE_REVISION "0000000000000000000000000000000000000000")
endif()

target_compile_definitions(acc PRIVATE
  ACC_SDK_PRODUCT_VERSION="${ACC_PRODUCT_VERSION}"
  ACC_SDK_SOURCE_REVISION="${ACC_SOURCE_REVISION}"
)
target_link_libraries(acc PRIVATE
  AgenticCircuit::ProjectOptions
  ACIRCodeGen
  ACIRDialect
  MLIRParser
)
