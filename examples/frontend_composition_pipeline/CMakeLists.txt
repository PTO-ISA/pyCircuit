cmake_minimum_required(VERSION 3.25)
project(PycircuitFrontendCompositionPipeline LANGUAGES CXX)
find_package(pycircuit CONFIG REQUIRED COMPONENTS Runtime)
include("${PYCIRCUIT_CMAKE_DIR}/PycircuitExamples.cmake")
pycircuit_add_example(frontend_composition_pipeline
  PACKAGE example_frontend_composition_pipeline
  TOP example_frontend_composition_pipeline.frontend_composition_pipeline.FrontendCompositionPipeline
  ENTRY_SOURCE frontend_composition_pipeline
  SOURCES frontend_composition_pipeline.py)
if(PYC_EXAMPLE_VERIFY)

find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(IVERILOG_EXECUTABLE iverilog REQUIRED)
find_program(VVP_EXECUTABLE vvp REQUIRED)
add_test(NAME frontend_composition_pipeline_four_state
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/four_state.py"
    --source "${CMAKE_CURRENT_SOURCE_DIR}"
    --build "${CMAKE_CURRENT_BINARY_DIR}"
    --include "${PYCIRCUIT_RUNTIME_INCLUDE_DIR}"
    --scratch "${CMAKE_CURRENT_BINARY_DIR}/four-state"
    --iverilog "${IVERILOG_EXECUTABLE}"
    --vvp "${VVP_EXECUTABLE}")
set_tests_properties(frontend_composition_pipeline_four_state PROPERTIES DEPENDS frontend_composition_pipeline)

endif()
