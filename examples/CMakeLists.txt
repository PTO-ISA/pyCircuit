cmake_minimum_required(VERSION 3.25)
project(PycircuitExamples LANGUAGES CXX)
enable_testing()

set(PYC_EXAMPLE_TIER "nightly" CACHE STRING "Example selection: gate or nightly")
set_property(CACHE PYC_EXAMPLE_TIER PROPERTY STRINGS gate nightly)
if(NOT PYC_EXAMPLE_TIER MATCHES "^(gate|nightly)$")
  message(FATAL_ERROR "PYC_EXAMPLE_TIER must be gate or nightly")
endif()
# The catalog owns both retained designs and the current execution set.
# Pending designs keep their source/oracles and standalone CMake entrypoints.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS catalog.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/catalog.json" _catalog)
string(JSON _count LENGTH "${_catalog}" examples)
math(EXPR _last "${_count} - 1")
set(_selected_count 0)
foreach(_index RANGE ${_last})
  string(JSON _name GET "${_catalog}" examples ${_index} name)
  string(JSON _tier GET "${_catalog}" examples ${_index} execution_tier)
  string(JSON _status GET "${_catalog}" examples ${_index} migration status)
  if(NOT _name MATCHES "^[a-z][a-z0-9_]*$" OR
     NOT _tier MATCHES "^(gate|nightly|none)$")
    message(FATAL_ERROR "Invalid example selection in catalog: ${_name}")
  endif()
  if(_tier STREQUAL "none")
    continue()
  endif()
  if(NOT _status STREQUAL "migrated")
    message(FATAL_ERROR "Executable example is not migrated: ${_name}")
  endif()
  if(_tier STREQUAL "gate")
    set(PYC_EXAMPLE_LABELS "examples;gate;nightly")
  else()
    if(PYC_EXAMPLE_TIER STREQUAL "gate")
      continue()
    endif()
    set(PYC_EXAMPLE_LABELS "examples;nightly")
  endif()
  add_subdirectory(${_name})
  math(EXPR _selected_count "${_selected_count} + 1")
endforeach()
if(_selected_count EQUAL 0)
  message(FATAL_ERROR "The selected example execution set is empty")
endif()
